flowchart TB
  subgraph MGMT["管理界面层"]
    direction LR

  end
  GW -- "端口通过环境变量传递<br/>(ZLM_RTMP_PORT, ZLM_HTTP_PORT)" --> ZLM
  subgraph ZLM["数据面 (ZLMediaKit)"]
    direction LR
    Z1["RTMP :1935"]
    Z2["HTTP-FLV :8080"]
    Z3["HLS :8080"]
    Z4["RTSP :8554"]
  end

    R3["run/zlm.runtime.ini<br/>动态生成的 ZLM 配置"]

  end
